```ts
// lib/carousel/tiers.js
export const TIERS: { desktop: Tier; mobile: Tier; low: Tier };
// Tier = { name, dprCap, fboScale, samples, blur, aniso, continuous, ring: "720"|"1280" }
// desktop {2,1,16,true,16,true,"1280"} mobile {1.5,1,8,false,4,false,"720"} low {1.25,0.75,6,false,2,false,"720"}
export function detectTier(s: { override?, reducedMotion, saveData, webgl2, coarse, width }): "desktop"|"mobile"|"low"|"grid";
export function fpsVerdict(tierName, p50ms): "desktop"|"mobile"|"low"|"grid"; // >50 → grid; >36 && !low → low; else same
export function parseOverrides(search: string): { tier: TierName|"grid"|null; probe: boolean; entry: boolean; debug: "fps"|null }; // allow-list only
// lib/carousel/geometry.js
export function panelHeight(W, H, tierName): number;   // desktop clamp(320, round(0.5H), 620); mobile/low clamp(260, round(0.43H), 420)
export function computeRepeats(totalWidth, W, buffer): number; // max(3, ceil((W + 2*buffer)/totalWidth) + 1)
export function focusFit(aspect, W, H, tierName): { x, y, w, h, fit: "contain"|"cover" };
//   desktop: h = H-120, w = h*aspect; if w > W-192 → w = W-192, h = w/aspect; centred; fit "contain"
//   mobile/low: if aspect < 1 and visible fraction ((W/H)/aspect) >= 0.8 → full viewport rect, fit "cover" (D-13); else contain-fit in W×H
export function toScreenX(x, dir): number;               // rtl → -x
export function wheelDelta(e: { deltaX, deltaY, deltaMode }, dir): number; // deltaMode 1 → ×16; logical = deltaY + deltaX*sgn
export function gaussWeights(n): Array<[number, number, number]>; // t=i/(n-1): exp(-((t-0)/.38)^2), exp(-((t-.5)/.38)^2), exp(-((t-1)/.38)^2)
// lib/carousel/engine.d.ts (this plan's subset)
export class CarouselInitError extends Error {}
export type EngineItem = { id: string; aspect: number; src: string };
export type Rect = { x: number; y: number; w: number; h: number };
export type CarouselOptions = { items: EngineItem[]; tier: Tier; dir?: "ltr"|"rtl"; startIndex?: number; playEntry?: boolean; cursorElement?: HTMLElement|null;
  onActiveChange?(i: number): void; onSettle?(i: number): void; onEntryDone?(): void; onFocusChange?(open: boolean): void };
export type CarouselHandle = { closeFocus(): void; getCentredRect(): Rect|null; pause(): void; resume(): void; destroy(): void; lensUniforms: Record<string, { value: unknown }> };
export function createCarousel(mount: HTMLElement, opts: CarouselOptions): CarouselHandle;
```
